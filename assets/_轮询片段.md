# 轮询片段（共享）

> 本文档是待命/休眠轮询代码的**唯一权威来源**。
> 角色模板和各模式文件均引用此处——修改一轮询逻辑，全项目生效。
> 内含优化：信号文件检测、去重复 import、心跳批量写入、持久化轮询日志。

---

## 使用说明

在引用本文档的位置（角色模板、各模式文件的「本轮后」部分），根据公告牌指定的状态选择对应片段，替换 `{{ROLE_NAME}}`（scaffold 已在实际 AGENTS.md 中替换为具体角色名）。

> ⚠️ 角色第一次执行前应先读自己的 AGENTS.md 确认 `{{ROLE_NAME}}` 已替换为实际角色名。

---

## 休眠轮询（sleepPoll）

```js
// === 休眠轮询 ===
// 特性：O(1)直查 + mtime预检 + 信号文件检测 + 去import + 心跳批量写入 + 240s自愈 + 持久化日志
var _fs = await import("node:fs");
var _lastN = 0;
var _pollCount = 0;
var _hbCount = 0;
var _startTime = Date.now();
var _dirMtime = 0;
var _logRole = null;
function _log(msg) {
  try {
    if (!_logRole) {
      try { var _ag = _fs.default.readFileSync("./AGENTS.md", "utf8"); var _rm = _ag.match(/^# (.+)$/m); _logRole = (_rm ? _rm[1].trim() : "unknown"); } catch(_e2) { _logRole = "unknown"; }
    }
    var _logDir = "../我的世界/" + _logRole + "_大鱼对讲";
    _fs.default.mkdirSync(_logDir, { recursive: true });
    var _ts = new Date().toISOString().substring(11, 19);
    _fs.default.appendFileSync(_logDir + "/" + _logRole + "_轮询日志.md", "[" + _ts + "] " + msg + "\n", "utf8");
  } catch(_e3) {}
}
if (_lastN <= 1) _log("=== 新轮次周期 N=" + (_lastN || 1) + " ===");
_log("轮询启动（休眠）N=" + _lastN);
while (true) {
  try {
    var _world = "../我的世界";
    // 1. 信号文件优先检测：大鱼写了 _round_NNN.signal → 立刻跳过mtime检查
    var _sigFile = _world + "/_round_" + String(_lastN + 1).padStart(3,"0") + ".signal";
    if (_fs.default.existsSync(_sigFile)) {
      _lastN++;
      if (_fs.default.existsSync(_world + "/公告牌_" + String(_lastN).padStart(3,"0") + ".md")) {
        var _sigMtime = new Date(_fs.default.statSync(_sigFile).mtimeMs).toISOString().substring(11,19); _log("信号命中 N=" + _lastN + " → 公告牌_" + String(_lastN).padStart(3,"0") + " 就位（大鱼发布: " + _sigMtime + "）"); break;
      } }
    }
    // 2. 目录mtime预检：无变化则跳过文件检查（省statSync）
    var _dm = _fs.default.statSync(_world).mtimeMs;
    if (_dm !== _dirMtime) {
      _dirMtime = _dm;
      // O(1)直查：首次扫描确定当前N，后续只检查N+1
      if (_lastN === 0) {
        var _files = _fs.default.readdirSync(_world).filter(function(f){return f.startsWith("公告牌_")&&f.endsWith(".md")});
        _files.forEach(function(f){var n=parseInt(f.replace("公告牌_","").replace(".md",""));if(!isNaN(n)&&n>_lastN)_lastN=n;});
      }
      // 检查下一公告牌
      var _next = _world + "/公告牌_" + String(_lastN + 1).padStart(3,"0") + ".md";
      if (_fs.default.existsSync(_next)) { _lastN++; _log("公告牌_" + String(_lastN).padStart(3,"0") + " 就位"); break; }
      // 收工检查：当前公告牌是否为收工轮
      var _cur = _world + "/公告牌_" + String(_lastN).padStart(3,"0") + ".md";
      if (_fs.default.existsSync(_cur)) {
        var _bc = _fs.default.readFileSync(_cur, "utf8");
        if (/模式[：:]\s*收工/.test(_bc)) { _log("收工轮 N=" + _lastN); break; }
      }
      // 唤醒检查
      var _wake = _world + "/{{ROLE_NAME}}_大鱼对讲/_wakeup.md";
      if (_fs.default.existsSync(_wake)) { _fs.default.renameSync(_wake, _wake.replace('.md', '_acked.md')); _log("被唤醒"); break; }
    }
    // 3. 心跳批量写入：每5周期写一次（15s间隔），2min超时下安全边际8x
    _hbCount++;
    if (_hbCount % 5 === 0) {
      var _hb = "../我的世界/{{ROLE_NAME}}_大鱼对讲/_heartbeat.txt";
      _fs.default.writeFileSync(_hb + ".tmp", String(Date.now()), "utf8");
      _fs.default.renameSync(_hb + ".tmp", _hb);
    }
  } catch(_e) {}
  if (_pollCount % 20 === 0) console.log("[休眠轮询] " + new Date().toISOString() + " | N=" + _lastN + " (3s间隔)");
  await new Promise(function(r) { setTimeout(r, 3000); });
  _pollCount++;
  // REPL 240s 自愈：300s硬上限前主动退出，模型自然重发续命
  if (Date.now() - _startTime > 240000) { _log("240s自愈 已运行 " + Math.round((Date.now() - _startTime) / 1000) + "s"); break; }
}
```
